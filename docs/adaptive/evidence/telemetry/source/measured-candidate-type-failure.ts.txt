// Benchmark-only host. Uses the product runtime and MCP surface without new agent tools.
import { appendFileSync } from "node:fs";
import { hostSandbox } from "../../src/core/sandbox.ts";
import { FileStore, serveStdio } from "../../src/entry/node-host.ts";
import { RightClick } from "../../src/runtime.ts";
import { recordSandbox } from "./observation.ts";

const env = process.env;
for (const name of ["RCX_OBSERVATION_LOG", "RCX_SESSION_ID", "RCX_INVESTIGATE", "RCX_STATE_FILE", "RCX_MCP_LOG"]) if (!env[name]) throw new Error(`missing ${name}`);
const observation = recordSandbox(env.RCX_OBSERVATION_LOG!, env.RCX_SESSION_ID!, env.RCX_INVESTIGATE!, await hostSandbox(), "candidate");
process.once("exit", observation.close);
// Domain 1 exposes one program, no network or workspace filesystem capabilities. Measurement,
// oracle and state files must never become readable/writable through the candidate's tools.
const rc = await RightClick.create({ name: "rightclick-x", directories: [], network: { origins: [] }, store: new FileStore(env.RCX_STATE_FILE!), authority: { investigate: [env.RCX_INVESTIGATE!], execute: [env.RCX_INVESTIGATE!] }, sandbox: observation.sandbox, onInvestigationStart: observation.onInvestigationStart });
process.once("SIGTERM", () => { void rc.flush().finally(() => process.exit(0)); });
rc.toolCallSink = entry => appendFileSync(env.RCX_MCP_LOG!, JSON.stringify(entry) + "\n");
await serveStdio(rc);
