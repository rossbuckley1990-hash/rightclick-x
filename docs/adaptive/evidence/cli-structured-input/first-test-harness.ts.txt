import { test } from "node:test";
import assert from "node:assert/strict";
import { spawnSync } from "node:child_process";
import { existsSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { resolve } from "node:path";

const cli = resolve("bin/rcx.ts");
function fixture() {
  const dir = mkdtempSync(`${tmpdir()}/rcx-command-input-`);
  const state = `${dir}/state.json`;
  const call = (args: string[], stdin?: string, policy?: object) => spawnSync(process.execPath, [cli, ...args], {
    cwd: dir, env: { PATH: process.env.PATH, HOME: dir, RCX_STATE_FILE: state, RCX_WORKSPACE: dir,
      ...(policy ? { RCX_EXECUTION_POLICY: JSON.stringify(policy) } : {}) },
    input: stdin, encoding: "utf8", timeout: 15000,
  });
  return { dir, state, call, close: () => rmSync(dir, { recursive: true, force: true }) };
}

test("CLI structured arguments preserve arrays, nested values, Unicode and literal strings", () => {
  const f = fixture();
  try {
    const value = { items: [1, false, null, { text: "雪\n`$HOME`\"" }], empty: {}, exact: "9007199254740993" };
    // The built-in encoder provides an observable round trip, not a mocked parser result.
    const input = JSON.stringify({ value });
    const result = f.call(["run", "builtin:json.stringify", "--args", input]);
    assert.equal(result.status, 0, result.stderr);
    assert.deepEqual(JSON.parse(JSON.parse(result.stdout).output), value);
  } finally { f.close(); }
});

test("CLI accepts a JSON file and piped JSON in fresh processes", () => {
  const f = fixture();
  try {
    const path = `${f.dir}/input with spaces.json`;
    const args = { value: { files: { "a b.txt": "data\n", "__proto__": "literal" }, operands: [], options: ["-e", "print"] } };
    writeFileSync(path, JSON.stringify(args));
    for (const [flags, stdin] of [[["--args-file", path], undefined], [["--args-file", "-"], JSON.stringify(args)]] as const) {
      const result = f.call(["run", "builtin:json.stringify", ...flags], stdin);
      assert.equal(result.status, 0, result.stderr);
      assert.deepEqual(JSON.parse(JSON.parse(result.stdout).output), args.value);
    }
  } finally { f.close(); }
});

test("CLI compose accepts typed structured inputs and retains scalar argument spelling", () => {
  const f = fixture();
  try {
    const result = f.call(["compose", "--args", JSON.stringify({ value: { type: "text", value: "literal\ntext" } }), "--want", "hash", "--run"]);
    assert.equal(result.status, 0, result.stderr);
    assert.equal(JSON.parse(result.stdout).executed.status, "completed");
    const scalar = f.call(["run", "builtin:json.stringify", "value=00123"]);
    assert.equal(scalar.status, 0, scalar.stderr);
    assert.equal(JSON.parse(JSON.parse(scalar.stdout).output), "00123");
  } finally { f.close(); }
});

test("ambiguous or lossy CLI arguments refuse before creating runtime state", () => {
  const invalid = [
    ["--args", "[]"], ["--args", "null"], ["--args", "{broken"], ["--args"],
    ["--args", '{"value":9007199254740993}'], ["--args", '{"value":1,"value":2}'],
    ["--args", "{}", "value=second"], ["--args", "{}", "--args", "{}"],
    ["--args", "{}", "--args-file", "-"], ["--args-file"],
    ["--unknown", "x"], ["value=one", "value=two"], ["not-an-assignment"],
  ];
  for (const args of invalid) {
    const f = fixture();
    try {
      const result = f.call(["run", "builtin:json.stringify", ...args], "{}");
      assert.notEqual(result.status, 0, JSON.stringify(args));
      assert.equal(result.stdout, "");
      assert.equal(existsSync(f.state), false, "bad input must not load, migrate or save runtime state");
    } finally { f.close(); }
  }
});

test("invalid file encoding and missing input files refuse before runtime state", () => {
  const f = fixture();
  try {
    const invalid = `${f.dir}/invalid.json`;
    writeFileSync(invalid, Buffer.from([0x7b, 0x22, 0x78, 0x22, 0x3a, 0x22, 0xff, 0x22, 0x7d]));
    for (const path of [invalid, `${f.dir}/missing.json`]) {
      const result = f.call(["run", "builtin:json.stringify", "--args-file", path]);
      assert.notEqual(result.status, 0);
      assert.equal(existsSync(f.state), false);
    }
  } finally { f.close(); }
});

test("CLI execution restrictions remain effective and failed runs exit unsuccessfully", () => {
  const f = fixture();
  try {
    const result = f.call(["run", "builtin:json.stringify", "--args", '{"value":{"authority":"allow"}}'], undefined, { allowedRisks: [] });
    assert.notEqual(result.status, 0);
    const run = JSON.parse(result.stdout);
    assert.equal(run.status, "failed");
    assert.match(run.error, /execution policy/);
    assert.equal(run.output, undefined);
    assert.ok(JSON.parse(readFileSync(f.state, "utf8")).receipts.length);
  } finally { f.close(); }
});
